#include <iostream>
#include <string>
#include <cctype>
using namespace std;

int main() {
    string parrafo;
    cout << "Ingrese un párrafo: ";
    getline(cin, parrafo);

    int letras = 0, numeros = 0, espacios = 0, simbolos = 0;
    int mayusculas = 0, palabras = 0;
    bool enPalabra = false;

    for (int i = 0; i < parrafo.length(); i++) {
        char c = parrafo[i];

        if (isalpha(c)) {
            letras++;
            if (isupper(c)) mayusculas++;
        }
        else if (isdigit(c)) numeros++;
        else if (isspace(c)) espacios++;
        else if (ispunct(c)) simbolos++;

        if (isspace(c)) {
            enPalabra = false;
        } else if (!enPalabra) {
            palabras++;
            enPalabra = true;
        }
    }

    double porcentaje = 0;
    if (letras > 0) porcentaje = mayusculas * 100.0 / letras;

    cout << "Letras: " << letras << endl;
    cout << "Números: " << numeros << endl;
    cout << "Espacios: " << espacios << endl;
    cout << "Símbolos: " << simbolos << endl;
    cout << "Palabras: " << palabras << endl;
    cout << "Porcentaje de mayúsculas: " << porcentaje << "%" << endl;

    return 0;
}

    for (int i = 0; i < oracion.length(); i++) {
        if (isalpha(oracion[i])) letras++;

        if (isspace(oracion[i])) {
            enPalabra = false;
        } else if (!enPalabra) {
            palabras++;
            enPalabra = true;
        }
    }

    cout << "Letras: " << letras << endl;
    cout << "Palabras: " << palabras << endl;
    return 0;
}

#include <iostream>
#include <string>
#include <cctype>
using namespace std;

int main() {
    string frase, limpia = "";
    cout << "Ingrese una palabra o frase: ";
    getline(cin, frase);

    // Quedarse solo con letras y números, en minúsculas
    for (int i = 0; i < frase.length(); i++) {
        if (isalnum(frase[i]))
            limpia += tolower(frase[i]);
    }

    bool palindromo = true;
    int inicio = 0, fin = limpia.length() - 1;

    while (inicio < fin) {
        if (limpia[inicio] != limpia[fin]) {
            palindromo = false;
            break;
        }
        inicio++;
        fin--;
    }

    if (palindromo)
        cout << "Es palíndromo" << endl;
    else
        cout << "No es palíndromo" << endl;

    return 0;
}

int main() {
    const int N = 5;
    string palabras[N];

    for (int i = 0; i < N; i++) {
        cout << "Palabra " << i + 1 << ": ";
        getline(cin, palabras[i]);
    }

    // Burbuja
    for (int i = 0; i < N - 1; i++) {
        for (int j = 0; j < N - 1 - i; j++) {
            if (palabras[j].compare(palabras[j + 1]) > 0) {
                string aux = palabras[j];
                palabras[j] = palabras[j + 1];
                palabras[j + 1] = aux;
            }
        }
    }

    cout << "\nOrdenadas:" << endl;
    for (int i = 0; i < N; i++)
        cout << palabras[i] << endl;

    return 0;
}

#include <iostream>
#include <string>
#include <cctype>
using namespace std;

string aMinusculas(string texto) {
    for (int i = 0; i < texto.length(); i++)
        texto[i] = tolower(texto[i]);
    return texto;
}

string aMayusculas(string texto) {
    for (int i = 0; i < texto.length(); i++)
        texto[i] = toupper(texto[i]);
    return texto;
}

int contarVocales(string texto) {
    int cont = 0;
    for (int i = 0; i < texto.length(); i++) {
        switch (tolower(texto[i])) {
            case 'a': case 'e': case 'i': case 'o': case 'u':
                cont++;
        }
    }
    return cont;
}

string invertir(string texto) {
    string resultado = "";
    for (int i = texto.length() - 1; i >= 0; i--)
        resultado += texto[i];
    return resultado;
}

int main() {
    string frase;
    int opcion;

    cout << "Ingrese una frase: ";
    getline(cin, frase);

    cout << "\n1. Convertir a minúsculas" << endl;
    cout << "2. Convertir a mayúsculas" << endl;
    cout << "3. Contar vocales" << endl;
    cout << "4. Invertir texto" << endl;
    cout << "Elija una opción: ";
    cin >> opcion;

    switch (opcion) {
        case 1:
            cout << aMinusculas(frase) << endl;
            break;
        case 2:
            cout << aMayusculas(frase) << endl;
            break;
        case 3:
            cout << "Vocales: " << contarVocales(frase) << endl;
            break;
        case 4:
            cout << invertir(frase) << endl;
            break;
        default:
            cout << "Opción no válida" << endl;
    }

    return 0;
}
